cmake_minimum_required(VERSION 3.20)
project(faultdebug_install_consumer LANGUAGES C CXX)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

if(NOT DEFINED FAULTDEBUG_CMAKE_DIR)
  message(FATAL_ERROR "FAULTDEBUG_CMAKE_DIR must point to the installed CMake modules")
endif()
if(NOT DEFINED FAULTDEBUG_RUNTIME_DIR)
  message(FATAL_ERROR "FAULTDEBUG_RUNTIME_DIR must point to the installed runtime library")
endif()
include("${FAULTDEBUG_CMAKE_DIR}/FaultDebugLegacy.cmake")

add_executable(fd_install_c c_consumer.c)
add_executable(fd_install_cpp cpp_consumer.cpp)
foreach(target fd_install_c fd_install_cpp)
  faultdebug_instrument_existing(${target})
  target_link_options(${target} PRIVATE -pie)
endforeach()
